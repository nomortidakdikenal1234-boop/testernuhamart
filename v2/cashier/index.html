Diagnosis Tabel orders_v2 Supabase

Jika timbul pesan error pada koneksi, jalankan perintah SQL berikut di **Supabase SQL Editor** untuk memastikan tabel khusus orders_v2 terkonfigurasi dengan benar: 

-- 1. Buat Tabel orders_v2 khusus V2 jika belum ada
CREATE TABLE IF NOT EXISTS public.orders_v2 (
    id BIGSERIAL PRIMARY KEY,
    order_id TEXT UNIQUE,
    source TEXT DEFAULT 'Web V2',
    name TEXT,
    class_name TEXT,
    room TEXT,
    payment TEXT DEFAULT 'Cash / Tunai',
    status TEXT DEFAULT 'Pending',
    total_price NUMERIC DEFAULT 0,
    total_weight NUMERIC DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    items JSONB DEFAULT '[]'::jsonb,
    items_summary TEXT,
    cash_received NUMERIC DEFAULT 0,
    cash_change NUMERIC DEFAULT 0
);

-- 2. Matikan RLS agar Anon/Publik bisa Insert & Read tanpa hambatan
ALTER TABLE public.orders_v2 DISABLE ROW LEVEL SECURITY;

-- 3. Beri Izin Akses Penuh ke Peran Anon (Publik)
GRANT ALL ON TABLE public.orders_v2 TO anon, authenticated, service_role;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO anon, authenticated, service_role;

-- 4. Aktifkan Realtime Publikasi
ALTER PUBLICATION supabase_realtime ADD TABLE public.orders_v2;
